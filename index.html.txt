  <p>Kataklarni surib, 1 dan 8 gacha ketma-ket joylashtiring!</p>

  <div id="win-msg">🎉 Tabriklaymiz! Puzzle muvaffaqiyatli yechildi!</div>

  <button class="btn-restart" onclick="initGame()">Qayta boshlash 🔄</button>

  <script>
    const tg = window.Telegram.WebApp;
    tg.expand(); // Mini App-ni to'liq ekranga yoyish

    let boardState = [1, 2, 3, 4, 5, 6, 7, 8, ""];

    function shuffle(array) {
      for (let i = array.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [array[i], array[j]] = [array[j], array[i]];
      }
    }

    function renderBoard() {
      const boardEl = document.getElementById("board");
      boardEl.innerHTML = "";

      boardState.forEach((val, idx) => {
        const tile = document.createElement("div");
        tile.className = "tile" + (val === "" ? " empty" : "");
        tile.innerText = val;
        tile.onclick = () => moveTile(idx);
        boardEl.appendChild(tile);
      });
    }

    function moveTile(idx) {
      const emptyIdx = boardState.indexOf("");
      const row1 = Math.floor(idx / 3), col1 = idx % 3;
      const row2 = Math.floor(emptyIdx / 3), col2 = emptyIdx % 3;

      // Yonma-yon kataklarni surish
      if (Math.abs(row1 - row2) + Math.abs(col1 - col2) === 1) {
        [boardState[idx], boardState[emptyIdx]] = [boardState[emptyIdx], boardState[idx]];
        renderBoard();
        checkWin();
      }
    }

    function checkWin() {
      const winState = [1, 2, 3, 4, 5, 6, 7, 8, ""];
      const isWin = boardState.every((val, i) => val === winState[i]);

      if (isWin) {
        document.getElementById("win-msg").style.display = "block";
        tg.HapticFeedback.notificationOccurred('success');
      }
    }

    function initGame() {
      document.getElementById("win-msg").style.display = "none";
      shuffle(boardState);
      renderBoard();
    }

    initGame();
  </script>
